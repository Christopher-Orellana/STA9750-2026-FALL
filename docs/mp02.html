get_eia_sep <- function(state, abbr){
    library(tidyverse)
    library(stringr)
    library(httr2)
    library(rvest)
    library(fs)
  
    state_formatted <- str_to_lower(state) |> str_replace_all("\\s", "")
    
    dir_name <- fs::path("data", "mp02")
    file_name <- fs::path(dir_name, state_formatted)
    
    fs::dir_create(dir_name, recurse=TRUE)
    
    if(!fs::file_exists(file_name)){
        BASE_URL <- "https://www.eia.gov"
        RESPONSE <- request(BASE_URL) |> 
          req_url_path("electricity", 
                       "state", 
                       state_formatted) |>
          req_headers(
            `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0"
            ) |>   
          req_retry(
                max_tries = 4, 
          ) |>
          req_perform(file_name)
    
        resp_check_status(RESPONSE)
    }
    
    TABLE <- read_html(file_name) |> 
        html_element("table") |> 
        html_table() |>
        mutate(Item = str_to_lower(Item))
    
    if("U.S. rank" %in% colnames(TABLE)){
        TABLE <- TABLE |> rename(Rank = `U.S. rank`)
    }
    
    CO2_MWh <- TABLE |> 
        filter(Item == "carbon dioxide (lbs/mwh)") |>
        pull(Value) |> 
        str_replace_all(",", "") |>
        as.numeric()
    
    PRIMARY <- TABLE |> 
        filter(Item == "primary energy source") |> 
        pull(Rank)
    
    RATE <- TABLE |>
        filter(Item == "average retail price (cents/kwh)") |>
        pull(Value) |>
        as.numeric()
    
    GENERATION_MWh <- TABLE |>
        filter(Item == "net generation (megawatthours)") |>
        pull(Value) |>
        str_replace_all(",", "") |>
        as.numeric()
    
    data.frame(CO2_MWh               = CO2_MWh, 
               primary_source        = PRIMARY,
               electricity_price_MWh = RATE * 10, # / 100 cents to dollars &
               # * 1000 kWh to MWH 
               generation_MWh        = GENERATION_MWh, 
               state                 = state, 
               abbreviation          = abbr
    )
}

EIA_SEP_REPORT <- map2(state.name, state.abb, get_eia_sep) |> list_rbind()

load_ntd_energy <- function(){
  library(readxl)
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_ENERGY_FILE <- fs::path(DATA_DIR, "2025_ntd_energy.xlsx")

  if(!fs::file_exists(NTD_ENERGY_FILE)){
    library(httr2)
  
    request("https://www.transit.dot.gov/") |>
      req_url_path("sites",
                   "fta.dot.gov",
                   "files",
                   "2026-09",
                   "2025 Energy Consumption_260806.xlsx") |> 
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_ENERGY_FILE)
  }
  
  if(file.info(NTD_ENERGY_FILE)$size < 1e4){
        cat("I was unable to download the NTD Energy File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_energy.xlsx')` and try again.")
    }

  to_numeric_fill_0 <- function(x){
    x <- if_else(x == "-", NA, x)
    replace_na(as.numeric(x), 0)
  }

  read_excel(NTD_ENERGY_FILE) |> 
    select(-c(`Reporter Type`, 
              `Reporting Module`, 
              `Gallons of Other Fuel`, 
              `Other Fuel Description`)) |>
    mutate(across(-c(`Agency Name`, 
                     `Mode`,
                     `TOS`), 
                  to_numeric_fill_0)) |>
    group_by(`NTD ID`, `Mode`, `Agency Name`) |>
    summarize(across(where(is.numeric), sum), 
              .groups = "keep") |>
    mutate(ENERGY = sum(c_across(c(where(is.numeric))))) |>
    filter(ENERGY > 0) |>
    select(-ENERGY) |>
    ungroup()
}

NTD_ENERGY <- load_ntd_energy()

load_ntd_service <- function(){
  library(httr2)
  library(tidyverse)
  library(fs)
  
  # Create 'data/mp02' directory if not already present
  DATA_DIR <- fs::path("data", "mp02")
  fs::dir_create(DATA_DIR, recurse=TRUE)

  NTD_SERVICE_FILE <- fs::path(DATA_DIR, "2025_ntd_service.csv")

  if(!fs::file_exists(NTD_SERVICE_FILE)){
    library(httr2)
  
    request("https://data.transportation.gov/resource/4fir-qbim.csv") |>
      req_url_query(`$query`=paste(
        "SELECT  `_5_digit_ntd_id` AS `NTD_ID`", 
        "`max_agency` AS `AGENCY`",
        "`max_reporter_type` AS `reporter_type`",
        "`max_organization_type` AS `organization_type`",
        "`max_city` AS `city`",
        "`max_state` AS `state`",
        "`mode`",
        "`type_of_service`",
        "`max_primary_uza_code` as `uza_code`",
        "`max_primary_uza_name` as `uza_name`",
        "`max_primary_uza_area_sq_miles` as `uza_sq_miles`",
        "`max_primary_uza_population` as `uza_population`",
        "`max_service_area_sq_miles` as `service_area_sq_miles`",
        "`max_service_area_population` as `service_area_population`",
        "`questionable_record` as `flagged_likely_inaccurate`",
        "`sum_unlinked_passenger_trips_upt` as `total_upt`",
        "`sum_passenger_miles` WHERE caseless_one_of(`report_year`, '2025') limit 50000", 
        sep=",")) |>
      req_headers(
        `User-Agent` = "Mozilla/5.0 (Macintosh; Intel Mac OS X 10.15; rv:149.0) Gecko/20100101 Firefox/149.0") |> 
      req_perform(NTD_SERVICE_FILE)
  }
  
  if(file.info(NTD_SERVICE_FILE)$size < 1e4){
        cat("I was unable to download the NTD Service File. Please try again.\n")
        stop("Download failed. Please delete the corrupted file by running `unlink('data/mp02/2025_ntd_service.xlsx')` and try again.")
    }

  read_csv(NTD_SERVICE_FILE) |>
    mutate(flagged_likely_inaccurate = !is.na(flagged_likely_inaccurate))
}

NTD_SERVICE <- load_ntd_service()

fuel_factors <- tribble(
  ~pattern, ~kg_co2, ~basis,
  "Bio-Diesel", 10.19, "Diesel proxy for unspecified blends",
  "Bunker", 11.24, "EIA residual fuel oil",
  "Compressed Natural", 52.91 * 0.138, "Natural gas; assumed diesel-equivalent gallon",
  "Diesel Fuel", 10.19, "EIA distillate fuel oil",
  "Ethanol", 0, "Biogenic CO2 excluded",
  "Gasoline", 8.49, "EIA finished motor gasoline",
  "Hydrogen", 0, "No direct carbon emissions; upstream production excluded",
  "Kerosene", 9.88, "EIA kerosene",
  "Liquified Nat|Liquefied Nat", 52.91 * 0.0843, "Natural gas; assumed 0.0843 MMBtu per liquid gallon",
  "Liquified Petroleum|Liquefied Petroleum", 5.75, "EIA propane proxy"
)
fuel_factor_lookup <- tibble(fuel = fuel_columns) |>
  mutate(kg_co2 = map_dbl(fuel, function(name) {
    if (str_detect(name, "Kilowatt")) return(0)
    match <- which(str_detect(name, fuel_factors$pattern))
    stopifnot(length(match) == 1L)
    fuel_factors$kg_co2[match]
  }))

  left_join(fuel_factor_lookup, by = "fuel", relationship = "many-to-one") |>

            electric_kwh = sum(quantity[str_detect(fuel, "Kilowatt")]),
            non_electric_use = any(quantity[!str_detect(fuel, "Kilowatt")] > 0),

agency_mode <- service_modes |>
  left_join(NTD_ENERGY, by = c("ntd_id", "mode"), relationship = "one-to-one") |>
  left_join(EIA_SEP_REPORT |> select(abbreviation, CO2_MWh),
            by = c("state" = "abbreviation"), relationship = "many-to-one") |>
  left_join(combustion, by = c("ntd_id", "mode"), relationship = "one-to-one") |>
  mutate(electric_kg = if_else(electric_kwh == 0, 0,
                              electric_kwh / 1000 * CO2_MWh * 0.45359237),
         emissions_kg = combustion_kg + electric_kg,
         kg_upt = emissions_kg / upt, kg_pmt = emissions_kg / pmt,
         complete = replace_na(!flagged & upt > 0 & pmt > 0 &
           emissions_kg > 0 & state %in% state.abb & !ntd_id %in% unmatched_ids, FALSE))

NEW YORK · November 3, 2026 — The Green Transit Alliance announces its 2026 Green Transit Awards. These awards recognize agencies that move passengers with low estimated CO₂ emissions. I used the assignment’s transit energy, service, and state electricity data to compare 401 eligible agencies.

Adirondack Transit Lines, Inc., dba: Adirondack Trailways; City of Seattle, dba: Seattle Center Monorail; Port Authority Trans-Hudson Corporation win Greenest Transit Agency in their size categories. Port Authority Trans-Hudson Corporation wins my original award, Electric Service Leader, for the share of passenger travel on fully electric modes.

I chose two awards, including one of my own, as allowed by the updated instructions. I also answered all eight questions in Tasks 3a and 3b.

Greenest Transit Agency

This award goes to the agency with the lowest estimated CO₂ per passenger-mile in each size group. This shows how much CO₂ an agency produces to move one passenger one mile. I grouped agencies by annual UPT: small means 100,000 to under 1 million, medium means 1 million to under 10 million, and large means 10 million or more. I chose these groups for the awards; they are not official NTD categories.

Electric Service Leader

Port Authority Trans-Hudson Corporation wins Electric Service Leader. Fully electric modes account for 100.0% of its passenger-miles, compared with 0.0% for the median eligible agency. That is 298,848,234 passenger-miles on fully electric modes.

For this award, I divided passenger-miles on fully electric modes by the agency’s total passenger-miles. A mode counts as fully electric only if it reports electricity and no other fuel. I left out mixed-fuel modes from the electric total because the data do not show which passengers used each fuel. If agencies tie, the one with more total passenger-miles wins, followed by the lower NTD ID.

Calculating emissions: Tasks 5 and 6

For each agency and mode, I multiplied fuel use by its CO₂ conversion factor. I calculated electricity emissions with this formula:

\[E_{\mathrm{electric}} = \frac{\mathrm{kWh}}{1000}\times \mathrm{state\ lb\ CO_2/MWh}\times 0.45359237.\]

Dividing by 1,000 changes kWh to MWh. Multiplying by 0.45359237 changes pounds to kilograms. I included both battery charging and electric propulsion. Then I added each agency’s emissions across its modes and divided by its total UPT and passenger-miles. I used the agency totals rather than averaging the individual mode ratios.

Award rules and limits: Tasks 7 and 8

To avoid giving awards based on very small amounts of service, I required at least 100,000 annual UPT and 100,000 passenger-miles. Agencies also need positive service and emissions for every joined mode, no questionable service flags, no unmatched agency–mode records, and a location in the 50-state sample. The lowest CO₂ per passenger-mile wins within each size group. If efficiency values tie, the lower NTD ID wins.

The provided energy function removes “other fuel” and changes missing fuel values to zero. This means the estimates can miss some emissions. I used each state’s average electricity emissions as instructed, even if an agency has a green-energy agreement. The results cover fuel use and electricity generation, not the full impact of building vehicles, construction, or fuel production.

I focused on Greenest Transit Agency and Electric Service Leader. The updated instructions allow two award types, so I did not add the optional Most Emissions Avoided or “worst of” awards.

Overall, these awards highlight two things: moving passengers with low estimated emissions and providing electric service. Looking at both gives a clearer picture than looking at fuel use alone.
